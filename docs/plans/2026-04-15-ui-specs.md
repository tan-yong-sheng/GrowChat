# Short-term Tasks — UI Specifications Plan

> **REQUIRED SUB-SKILL:** Use the executing-plans skill to implement this plan task-by-task.

**Goal:** Complete frontend UI for 4 short-term features: Email Verification, Message Editing, Session Management, and Audit Logs.

**Architecture:** Follow DESIGN.md patterns — My Settings uses drawer/sheet (personal scope), Admin Settings uses full-page workspace. Reuse existing components (settings shells, message actions, edit-textarea).

**Tech Stack:** Vanilla JS, Tailwind CSS, existing component library (`public/js/shared/components/`)

---

## Design Principles (from DESIGN.md)

1. **My Settings** = drawer/sheet over main app (personal scope)
2. **Admin Settings** = full-page workspace (admin scope)
3. **Shared components** underneath — one backend service, one capability matrix, one shared component set
4. **Scope should be visible** in page chrome, not hidden in route names

      resendBtn.textContent = 'Resend email';
      cooldownText.style.display = 'none';
    }
  }, 1000);

  resendBtn.addEventListener('click', () => onResend());
  container.querySelector('#continue-btn').addEventListener('click', onContinue);

  return container;
}

// public/js/features/chat/chat-message-edit.js
import { setupEditTextarea } from './edit-textarea.js';

export class MessageEditController {
  constructor(messageEl, { messageId, originalContent, onSave, onCancel }) {
    this.messageEl = messageEl;
    this.messageId = messageId;
    this.originalContent = originalContent;
    this.onSave = onSave;
    this.onCancel = onCancel;
    this.isEditing = false;
    this.contentEl = messageEl.querySelector('.message-content');
  }

  showEditButton() {
    const btn = this.messageEl.querySelector('.edit-btn');
    if (btn) btn.style.display = '';
  }

  hideEditButton() {
    const btn = this.messageEl.querySelector('.edit-btn');
    if (btn) btn.style.display = 'none';
  }

  enterEditMode() {
    if (this.isEditing) return;
    this.isEditing = true;

    this.originalHtml = this.contentEl.innerHTML;
    const currentText = this.contentEl.textContent;

    this.contentEl.innerHTML = `
      <textarea class="edit-textarea w-full p-2 border rounded resize-none">${escapeHtml(currentText)}</textarea>
      <div class="edit-actions mt-2 flex gap-2">
        <button class="save-edit-btn btn-primary btn-sm">Save</button>
        <button class="cancel-edit-btn btn-secondary btn-sm">Cancel</button>
      </div>
    `;

    const textarea = this.contentEl.querySelector('textarea');
    setupEditTextarea(textarea);

    textarea.focus();

    this.contentEl.querySelector('.save-edit-btn').addEventListener('click', () => this.save());
    this.contentEl.querySelector('.cancel-edit-btn').addEventListener('click', () => this.cancel());

    textarea.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        e.preventDefault();
        this.cancel();
      }
      if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
        e.preventDefault();
        this.save();
      }
    });
  }

  async save() {
    const textarea = this.contentEl.querySelector('textarea');
    const newContent = textarea.value.trim();

    if (!newContent) return;

    const saveBtn = this.contentEl.querySelector('.save-edit-btn');
    saveBtn.disabled = true;
    saveBtn.innerHTML = '<span class="spinner"></span> Saving...';

    try {
      await this.onSave(this.messageId, newContent);
      this.exitEditMode(true);
      this.showEditedBadge();
    } catch (err) {
      saveBtn.disabled = false;
      saveBtn.textContent = 'Save';
      // Show toast error
    }
  }

  cancel() {
    this.exitEditMode(false);
    this.onCancel?.();
  }

  exitEditMode(saved) {
    this.isEditing = false;
    if (saved) {
      const textarea = this.contentEl.querySelector('textarea');
      this.contentEl.innerHTML = escapeHtml(textarea.value);
    } else {
      this.contentEl.innerHTML = this.originalHtml;
    }
  }

  showEditedBadge() {
    const badge = document.createElement('span');
    badge.className = 'edited-badge text-xs text-gray-400 ml-2';
    badge.textContent = '(edited)';
    badge.title = new Date().toLocaleString();
    this.contentEl.appendChild(badge);
  }
}

export function createMessageEditController(messageEl, options) {
  return new MessageEditController(messageEl, options);
}
```

// public/js/features/account/sessions.js
export function renderSessionsList(sessions, { onRevoke }) {
  const container = document.createElement('div');
  container.className = 'sessions-list';

  const currentSession = sessions.find(s => s.isCurrent);
  const otherSessions = sessions.filter(s => !s.isCurrent);

  let html = '<h3 class="text-lg font-medium mb-4">Active Sessions</h3>';

  if (currentSession) {
    html += `
      <div class="session-card current-session bg-blue-50 border border-blue-200 rounded-lg p-4 mb-4" data-current="true">
        <div class="flex items-center justify-between">
          <div>
            <span class="font-medium">${escapeHtml(currentSession.device)}</span>
            <span class="ml-2 text-xs bg-blue-600 text-white px-2 py-0.5 rounded">This device</span>
            <p class="text-sm text-gray-500 mt-1">${escapeHtml(currentSession.ip || 'Unknown location')}</p>
          </div>
        </div>
      </div>
    `;
  }

  if (otherSessions.length > 0) {
    html += '<h4 class="text-sm font-medium text-gray-700 mb-2">Other sessions</h4>';
    otherSessions.forEach(session => {
      html += `
        <div class="session-card bg-white border rounded-lg p-4 mb-2" data-session-id="${session.id}">
          <div class="flex items-center justify-between">
            <div>
              <span class="font-medium">${escapeHtml(session.device)}</span>
              <p class="text-sm text-gray-500">${escapeHtml(session.ip || 'Unknown location')}</p>
              <p class="text-xs text-gray-400">Last active: ${formatRelativeTime(session.lastActive)}</p>
            </div>
            <button class="revoke-btn btn-danger btn-sm" aria-label="Revoke session from ${escapeHtml(session.device)}">
              Revoke
            </button>
          </div>
        </div>
      `;
    });

    html += `
      <button class="revoke-all-btn btn-danger mt-4" id="revoke-all-other">
        Revoke all other sessions
      </button>
    `;
  } else {
    html += '<p class="text-gray-500 text-sm">No other sessions</p>';
  }

  container.innerHTML = html;

  // Wire up revoke buttons
  container.querySelectorAll('.revoke-btn').forEach(btn => {
    btn.addEventListener('click', async (e) => {
      const card = e.target.closest('.session-card');
      const sessionId = card.dataset.sessionId;
      if (confirm('Are you sure you want to revoke this session?')) {
        await onRevoke(sessionId);
        card.remove();
      }
    });
  });

  const revokeAllBtn = container.querySelector('#revoke-all-other');
  if (revokeAllBtn) {
    revokeAllBtn.addEventListener('click', async () => {
      if (confirm('Are you sure you want to revoke all other sessions?')) {
        await Promise.all(otherSessions.map(s => onRevoke(s.id)));
        // Remove all other session cards
        container.querySelectorAll('.session-card:not([data-current])').forEach(c => c.remove());
        revokeAllBtn.remove();
        container.innerHTML += '<p class="text-gray-500 text-sm">No other sessions</p>';
      }
    });
  }

  return container;
}

function formatRelativeTime(timestamp) {
  if (!timestamp) return 'Unknown';
  const seconds = Math.floor((Date.now() / 1000) - timestamp);
  if (seconds < 60) return 'Just now';
  if (seconds < 3600) return `${Math.floor(seconds / 60)} minutes ago`;
  if (seconds < 86400) return `${Math.floor(seconds / 3600)} hours ago`;
  return `${Math.floor(seconds / 86400)} days ago`;
}

function escapeHtml(value) {
  return String(value ?? '').replaceAll('&', '&amp;').replaceAll('<', '&lt;').replaceAll('>', '&gt;');
}
```

// public/js/features/admin/audit-logs.js
export async function renderAuditLogsSection({ apiFetch }) {
  const container = document.createElement('div');
  container.className = 'audit-logs-section';

  let page = 1;
  let filters = { userId: '', action: '', dateFrom: '', dateTo: '' };

  async function loadLogs() {
    const params = new URLSearchParams({ limit: 50, offset: (page - 1) * 50 });
    if (filters.userId) params.append('userId', filters.userId);
    if (filters.action) params.append('action', filters.action);

    const res = await apiFetch(`/api/admin/audit-logs?${params}`);
    if (!res.ok) throw new Error('Failed to load audit logs');
    return res.json();
  }

  function renderFilters() {
    return `
      <div class="audit-filters flex gap-4 mb-4">
        <input type="text" id="filter-user" placeholder="Filter by user" class="input" value="${filters.userId}">
        <select id="filter-action" class="input">
          <option value="">All actions</option>
          <option value="auth.login">Login</option>
          <option value="auth.logout">Logout</option>
          <option value="user.update">User Update</option>
          <option value="admin.settings_change">Settings Change</option>
        </select>
        <button id="apply-filters" class="btn-primary">Apply</button>
        <button id="export-csv" class="btn-secondary">Export CSV</button>
      </div>
    `;
  }

  function renderTable(logs) {
    if (!logs.length) {
      return '<p class="text-gray-500">No audit logs found</p>';
    }

    return `
      <table class="audit-table w-full">
        <thead>
          <tr class="border-b">
